GASの見本ツール3つ

Googleのスプレッドシート・Gmailだけで動く、小さな自動化の見本です。新しいソフトを入れず、いつもの道具に「勝手に動く係」を足すイメージです。コードは全文を載せているので、そのままコピーして試せます。

これは「こういう仕組みを作れます」という見本です。お使いの環境に合わせた設計・設定のご相談は、トップページの案内からお願いします。

見本A：問い合わせメールを表に自動で記録

Gmailで「問い合わせ」ラベルを付けたメールを読み、日時・差出人・件名・本文の冒頭を、スプレッドシートに1通1行で書き足します。同じメールを二度書かないよう、メッセージIDで見分けます。15分ごとに自動で動くので、メールを見返して表に写す手間がなくなります。

使い方

  1. Gmailでラベル「問い合わせ」を作り、記録したいメールに付けます（フィルタで自動に付けることもできます）。
  2. 新しいスプレッドシートを作り、［拡張機能］→［Apps Script］を開きます。
  3. 下のコードをコピーして貼り付け、保存します。必要なら先頭の「設定」を書き換えます。
  4. 関数 setupTrigger を選んで［実行］します。権限の確認画面で許可すると、15分ごとの自動実行が登録されます。
  5. 試しに recordInquiries を実行して、シートに行が増えるのを確かめます。

コード全文 （sample-a.gs をダウンロード）

/**
 * 見本A：問い合わせメールを表に自動で記録
 * Gmailの特定ラベルが付いたメールを読み、スプレッドシートに1通1行で書き足します。
 * 同じメールは二度書きません（メッセージIDで判定）。
 *
 * 使い方：スプレッドシートの［拡張機能］→［Apps Script］に貼り付け、
 * 一度 setupTrigger を実行すると15分ごとに自動で動きます。
 */

// ===== 設定（ここだけ書き換えてください） =====
const CONFIG = {
  LABEL_NAME: '問い合わせ',       // 記録したいメールに付けるGmailのラベル名
  SHEET_NAME: '問い合わせ記録',   // 書き込むシート名（なければ自動で作ります）
  BODY_LENGTH: 200,               // 本文の冒頭を何文字まで記録するか
  MAX_THREADS: 50,                // 1回の実行で見るスレッド数の上限
};
// ============================================

const HEADERS = ['受信日時', '差出人', '件名', '本文（冒頭）', 'メッセージID'];
const ID_COLUMN = 5; // メッセージIDを入れる列（E列）

/** 15分ごとに動かす本体 */
function recordInquiries() {
  const lock = LockService.getScriptLock();
  if (!lock.tryLock(30000)) return; // 前の実行がまだ続いていたら今回は見送る
  try {
    const label = GmailApp.getUserLabelByName(CONFIG.LABEL_NAME);
    if (!label) {
      throw new Error('ラベル「' + CONFIG.LABEL_NAME + '」が見つかりません。Gmailで作成してください。');
    }

    const sheet = getSheet_();
    const lastRow = sheet.getLastRow();

    // すでに記録したメッセージIDの一覧
    const known = new Set();
    if (lastRow > 1) {
      sheet.getRange(2, ID_COLUMN, lastRow - 1, 1).getValues()
        .forEach(function (r) { known.add(String(r[0])); });
    }

    const newRows = [];
    label.getThreads(0, CONFIG.MAX_THREADS).forEach(function (thread) {
      thread.getMessages().forEach(function (msg) {
        const id = msg.getId();
        if (known.has(id)) return; // 記録済みは飛ばす
        known.add(id);
        const body = msg.getPlainBody().replace(/\s+/g, ' ').trim().slice(0, CONFIG.BODY_LENGTH);
        newRows.push([msg.getDate(), msg.getFrom(), msg.getSubject(), body, id]);
      });
    });

    if (newRows.length === 0) return;
    newRows.sort(function (a, b) { return a[0] - b[0]; }); // 古い順に並べる
    sheet.getRange(sheet.getLastRow() + 1, 1, newRows.length, HEADERS.length).setValues(newRows);
  } finally {
    lock.releaseLock();
  }
}

/** 書き込み先のシートを返す（なければ作って見出しを入れる） */
function getSheet_() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let sheet = ss.getSheetByName(CONFIG.SHEET_NAME);
  if (!sheet) sheet = ss.insertSheet(CONFIG.SHEET_NAME);
  if (sheet.getLastRow() === 0) {
    sheet.getRange(1, 1, 1, HEADERS.length).setValues([HEADERS]).setFontWeight('bold');
    sheet.setFrozenRows(1);
  }
  return sheet;
}

/** 最初に1回だけ実行：15分ごとのトリガーを登録する */
function setupTrigger() {
  ScriptApp.getProjectTriggers().forEach(function (t) {
    if (t.getHandlerFunction() === 'recordInquiries') ScriptApp.deleteTrigger(t);
  });
  ScriptApp.newTrigger('recordInquiries').timeBased().everyMinutes(15).create();
}
注意
  • 初回の実行で「このアプリは Google で確認されていません」などの画面が出ます。自分で貼ったスクリプトなので、内容を確認のうえ［詳細］から進んで許可します（Gmailの閲覧とスプレッドシートの編集の権限を求められます）。
  • メールの差出人や本文には個人情報が含まれます。表を共有する範囲は必要な人だけにし、本文は冒頭だけを記録する設定にしています。文字数は BODY_LENGTH で変えられます。
  • メッセージIDの列（E列）を消すと、同じメールがもう一度記録されます。

見本B：在庫が減ったら知らせる

スプレッドシートの在庫表（商品名・在庫数・下限）を毎朝読み、下限を下回った商品を1通のメールにまとめて知らせます。下回った商品がない日はメールを送りません。商品ごとに通知が何通も届くことがなく、朝に1通だけ確認すれば済みます。

使い方

  1. スプレッドシートに「在庫」シートを作り、1行目を見出し（商品名・在庫数・下限）、2行目から商品を入力します。
  2. ［拡張機能］→［Apps Script］を開き、下のコードを貼り付けて保存します。
  3. 先頭の「設定」で、宛先（NOTIFY_TO）と確認する時刻（HOUR）を決めます。宛先が空欄なら実行した本人に届きます。
  4. 関数 setupTrigger を選んで［実行］し、権限を許可すると、毎朝の自動実行が登録されます。
  5. 試しに在庫数を下限より小さくして checkStock を実行し、メールが届くのを確かめます。

コード全文 （sample-b.gs をダウンロード）

/**
 * 見本B：在庫が減ったら知らせる
 * 在庫表（商品名・在庫数・下限）を毎朝読み、下限を下回った商品を
 * まとめて1通のメールで知らせます。下回った商品がなければ送りません。
 *
 * 使い方：在庫表のあるスプレッドシートの［拡張機能］→［Apps Script］に貼り付け、
 * 一度 setupTrigger を実行すると毎朝自動で動きます。
 */

// ===== 設定（ここだけ書き換えてください） =====
const CONFIG = {
  SHEET_NAME: '在庫',      // 在庫表のシート名
  NOTIFY_TO: '',           // 通知の宛先メールアドレス。空欄ならこのスクリプトを実行する本人宛て
  HOUR: 8,                 // 毎朝何時ごろに確認するか（0〜23）
  // 列の位置（A列=1）。1行目は見出し、2行目から商品が並ぶ前提です
  COL_NAME: 1,             // 商品名
  COL_STOCK: 2,            // 在庫数
  COL_MIN: 3,              // 下限
};
// ============================================

/** 毎朝動かす本体 */
function checkStock() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName(CONFIG.SHEET_NAME);
  if (!sheet) throw new Error('シート「' + CONFIG.SHEET_NAME + '」が見つかりません。');

  const lastRow = sheet.getLastRow();
  if (lastRow < 2) return;
  const width = Math.max(CONFIG.COL_NAME, CONFIG.COL_STOCK, CONFIG.COL_MIN);
  const values = sheet.getRange(2, 1, lastRow - 1, width).getValues();

  const low = [];
  values.forEach(function (row) {
    const name = String(row[CONFIG.COL_NAME - 1]).trim();
    const stock = row[CONFIG.COL_STOCK - 1];
    const min = row[CONFIG.COL_MIN - 1];
    if (!name || stock === '' || min === '') return;        // 空欄の行は飛ばす
    if (isNaN(Number(stock)) || isNaN(Number(min))) return; // 数字でない行は飛ばす
    if (Number(stock) < Number(min)) {
      low.push('・' + name + '：在庫 ' + stock + '（下限 ' + min + '）');
    }
  });

  if (low.length === 0) return; // 問題なければ送らない

  const to = CONFIG.NOTIFY_TO || Session.getEffectiveUser().getEmail();
  const subject = '【在庫通知】下限を下回った商品が ' + low.length + ' 件あります';
  const body = '下限を下回っている商品は次のとおりです。\n\n' +
    low.join('\n') + '\n\n在庫表：' + ss.getUrl();
  MailApp.sendEmail(to, subject, body);
}

/** 最初に1回だけ実行：毎朝のトリガーを登録する */
function setupTrigger() {
  ScriptApp.getProjectTriggers().forEach(function (t) {
    if (t.getHandlerFunction() === 'checkStock') ScriptApp.deleteTrigger(t);
  });
  ScriptApp.newTrigger('checkStock').timeBased().everyDays(1).atHour(CONFIG.HOUR).create();
}
注意
  • 初回の実行で権限の確認画面が出ます。内容を確認して許可してください（スプレッドシートの閲覧とメール送信の権限を求められます）。
  • 宛先に社外のアドレスを入れると、在庫の情報が社外に送られます。宛先は必要な人に限ってください。
  • Googleアカウントの種類によって、1日に送れるメールの数に上限があります。この見本は1日1通までなので、通常は問題になりません。

見本C：注文メールから出荷用の表を作る

Gmailで「注文」ラベルを付けた注文通知メールを読み、注文番号・お名前・郵便番号・住所・電話番号・商品・数量を、スプレッドシートに1注文1行で書き足します。同じ注文は注文番号で見分け、二度書きません。読み取れなかった項目は空欄にして、「要確認」列に足りない項目名を書くので、そこだけ目で確かめれば済みます。あとから関数 exportCsv を実行すると、表をCSVにしてGoogleドライブに保存できます。

メール本文に「注文番号：」「お名前：」などの行がある形を例にしています。書き方が違う場合は、先頭の「設定」にある読み取りのパターン（正規表現）を書き換えます。

使い方

  1. Gmailでラベル「注文」を作り、注文通知メールに付けます（フィルタで自動に付けることもできます）。
  2. 新しいスプレッドシートを作り、［拡張機能］→［Apps Script］を開いて、下のコードを貼り付けて保存します。
  3. お手元の注文メールの書き方に合わせて、先頭の「設定」の PATTERNS を確かめます（保存先フォルダを決めるなら EXPORT_FOLDER_ID も）。
  4. 関数 setupTrigger を選んで［実行］し、権限を許可します。試しに recordOrders を実行して、シートに行が増えるのを確かめます。
  5. CSVが必要なときに関数 exportCsv を実行します。保存先のリンクは［実行ログ］に出ます。

コード全文 （sample-c.gs をダウンロード）

/**
 * 見本C：注文メールから出荷用の表を作る
 * Gmailの特定ラベルが付いた注文通知メールを読み、スプレッドシートに1注文1行で書き足します。
 * 同じ注文は二度書きません（注文番号で判定）。読み取れなかった項目は空欄にして「要確認」列に印を付けます。
 * 関数 exportCsv を実行すると、その表をCSVにしてGoogleドライブに保存します。
 *
 * 使い方：スプレッドシートの［拡張機能］→［Apps Script］に貼り付け、
 * 一度 setupTrigger を実行すると15分ごとに自動で動きます。
 *
 * 注意：氏名・住所・電話番号などの個人情報を扱います。表とCSVの共有範囲は必要な人だけにしてください。
 */

// ===== 設定（ここだけ書き換えてください） =====
const CONFIG = {
  LABEL_NAME: '注文',             // 記録したい注文メールに付けるGmailのラベル名
  SHEET_NAME: '出荷表',           // 書き込むシート名（なければ自動で作ります）
  MAX_THREADS: 50,                // 1回の実行で見るスレッド数の上限
  EXPORT_FOLDER_ID: '',           // CSVの保存先フォルダのID。空欄ならマイドライブの直下
  // 読み取りのパターン。メール本文の1行を読み、かっこ ( ) の中身を取り出します。
  // メールの書き方が違う場合は、ここだけ書き換えます（全角の「：」も半角の「:」も読めます）。
  PATTERNS: {
    orderNo: /^\s*注文番号\s*[：:]\s*(.+?)\s*$/m,
    name:    /^\s*お名前\s*[：:]\s*(.+?)\s*$/m,
    zip:     /^\s*郵便番号\s*[：:]\s*(.+?)\s*$/m,
    address: /^\s*住所\s*[：:]\s*(.+?)\s*$/m,
    phone:   /^\s*電話番号\s*[：:]\s*(.+?)\s*$/m,
    item:    /^\s*商品\s*[：:]\s*(.+?)\s*$/m,
    qty:     /^\s*数量\s*[：:]\s*(.+?)\s*$/m,
  },
};
// ============================================

// 表の見出し（読み取る項目の順番と、項目の表示名）
const FIELDS = [
  ['orderNo', '注文番号'],
  ['name', 'お名前'],
  ['zip', '郵便番号'],
  ['address', '住所'],
  ['phone', '電話番号'],
  ['item', '商品'],
  ['qty', '数量'],
];
const HEADERS = ['受信日時'].concat(
  FIELDS.map(function (f) { return f[1]; }),
  ['要確認', 'メッセージID']
);
const ORDER_COLUMN = 2;                 // 注文番号の列（B列）
const ID_COLUMN = HEADERS.length;       // メッセージIDの列（最後の列）
const TEXT_COLUMNS = [2, 4, 6];         // 先頭の0が消えないよう文字列で入れる列（注文番号・郵便番号・電話番号）

/** 15分ごとに動かす本体 */
function recordOrders() {
  const lock = LockService.getScriptLock();
  if (!lock.tryLock(30000)) return; // 前の実行がまだ続いていたら今回は見送る
  try {
    const label = GmailApp.getUserLabelByName(CONFIG.LABEL_NAME);
    if (!label) {
      throw new Error('ラベル「' + CONFIG.LABEL_NAME + '」が見つかりません。Gmailで作成してください。');
    }

    const sheet = getSheet_();
    const lastRow = sheet.getLastRow();

    // すでに記録した注文番号（注文番号が読めなかったメールはメッセージID）の一覧
    const known = new Set();
    if (lastRow > 1) {
      sheet.getRange(2, 1, lastRow - 1, HEADERS.length).getValues().forEach(function (r) {
        if (r[ORDER_COLUMN - 1] !== '') known.add(String(r[ORDER_COLUMN - 1]));
        known.add(String(r[ID_COLUMN - 1]));
      });
    }

    const newRows = [];
    label.getThreads(0, CONFIG.MAX_THREADS).forEach(function (thread) {
      thread.getMessages().forEach(function (msg) {
        const id = msg.getId();
        const parsed = parseOrder_(msg.getPlainBody());
        const key = parsed.orderNo || id;
        if (known.has(key) || known.has(id)) return; // 記録済みは飛ばす
        known.add(key);
        known.add(id);
        const missing = FIELDS
          .filter(function (f) { return parsed[f[0]] === ''; })
          .map(function (f) { return f[1]; });
        const flag = missing.length ? '要確認：' + missing.join('・') : '';
        const row = [msg.getDate()]
          .concat(FIELDS.map(function (f) { return parsed[f[0]]; }), [flag, id]);
        newRows.push(row);
      });
    });

    if (newRows.length === 0) return;
    newRows.sort(function (a, b) { return a[0] - b[0]; }); // 古い順に並べる
    const startRow = sheet.getLastRow() + 1;
    const range = sheet.getRange(startRow, 1, newRows.length, HEADERS.length);
    TEXT_COLUMNS.forEach(function (c) {
      sheet.getRange(startRow, c, newRows.length, 1).setNumberFormat('@'); // 文字列にして先頭の0を残す
    });
    sheet.getRange(startRow, 1, newRows.length, 1).setNumberFormat('yyyy/MM/dd HH:mm');
    range.setValues(newRows);
  } finally {
    lock.releaseLock();
  }
}

/** メール本文から各項目を読み取る（読めなかった項目は空欄） */
function parseOrder_(body) {
  const text = String(body).replace(/\r\n?/g, '\n');
  const result = {};
  FIELDS.forEach(function (f) {
    const m = CONFIG.PATTERNS[f[0]].exec(text);
    result[f[0]] = m ? m[1].trim() : '';
  });
  return result;
}

/** 書き込み先のシートを返す（なければ作って見出しを入れる） */
function getSheet_() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let sheet = ss.getSheetByName(CONFIG.SHEET_NAME);
  if (!sheet) sheet = ss.insertSheet(CONFIG.SHEET_NAME);
  if (sheet.getLastRow() === 0) {
    sheet.getRange(1, 1, 1, HEADERS.length).setValues([HEADERS]).setFontWeight('bold');
    sheet.setFrozenRows(1);
  }
  return sheet;
}

/** 表をCSVにしてGoogleドライブに保存する（必要なときに手で実行） */
function exportCsv() {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(CONFIG.SHEET_NAME);
  if (!sheet || sheet.getLastRow() < 2) throw new Error('書き出す注文がありません。');

  const tz = Session.getScriptTimeZone();
  const values = sheet.getDataRange().getValues();
  const lines = values.map(function (row) {
    return row.map(function (cell) {
      const s = cell instanceof Date ? Utilities.formatDate(cell, tz, 'yyyy/MM/dd HH:mm') : String(cell);
      return '"' + s.replace(/"/g, '""') + '"'; // 全項目を引用符で囲み、中の引用符は2つにする
    }).join(',');
  });

  // 先頭にBOMを付けると、Excelでも日本語が文字化けしにくい
  const csv = '﻿' + lines.join('\r\n');
  const name = '出荷表_' + Utilities.formatDate(new Date(), tz, 'yyyyMMdd_HHmm') + '.csv';
  const file = CONFIG.EXPORT_FOLDER_ID
    ? DriveApp.getFolderById(CONFIG.EXPORT_FOLDER_ID).createFile(name, csv, MimeType.CSV)
    : DriveApp.createFile(name, csv, MimeType.CSV);
  Logger.log('保存しました：' + file.getUrl());
}

/** 最初に1回だけ実行：15分ごとのトリガーを登録する */
function setupTrigger() {
  ScriptApp.getProjectTriggers().forEach(function (t) {
    if (t.getHandlerFunction() === 'recordOrders') ScriptApp.deleteTrigger(t);
  });
  ScriptApp.newTrigger('recordOrders').timeBased().everyMinutes(15).create();
}
注意
  • 氏名・住所・電話番号などの個人情報を扱います。表のシートと、保存したCSVのファイルは、共有する範囲を必要な人だけにしてください。CSVはドライブに残るので、使い終わったら削除するか、保存先フォルダの共有設定を確かめてください。
  • 初回の実行で権限の確認画面が出ます。内容を確認のうえ許可してください（Gmailの閲覧、スプレッドシートの編集、ドライブへのファイル作成の権限を求められます）。
  • 注文番号の列（B列）を消すと、同じ注文がもう一度記録されます。注文番号が読めなかったメールは、メッセージID（最後の列）で二重記録を防ぎます。
  • 読み取りは「項目名：内容」が1行に収まっている前提です。内容が複数行にわたる住所などは、途中までしか読めないことがあります。出荷の前に「要確認」列と表の中身を必ず見比べてください。

どれも、別々のスプレッドシート（別のスクリプト）に貼ってください。同じプロジェクトに両方を貼ると、設定の名前（CONFIG）などがぶつかります。
